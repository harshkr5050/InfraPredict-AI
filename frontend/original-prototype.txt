/* =================================================
   REGISTER
================================================= */

function register(){

    let name =
        document.getElementById("regName").value.trim();

    let mobile =
        document.getElementById("regMobile").value.trim();

    let email =
        document.getElementById("regEmail").value.trim().toLowerCase();

    let password =
        document.getElementById("regPassword").value;

    let confirm =
        document.getElementById("regConfirm").value;

    let msg =
        document.getElementById("registerMessage");


    if(!name || !mobile || !email || !password || !confirm){

        showMessage(
            msg,
            "Please fill all fields.",
            false
        );

        return;

    }


    if(!/^[0-9]{10}$/.test(mobile)){

        showMessage(
            msg,
            "Enter a valid 10 digit mobile number.",
            false
        );

        return;

    }


    if(password.length < 4){

        showMessage(
            msg,
            "Password must contain at least 4 characters.",
            false
        );

        return;

    }


    if(password !== confirm){

        showMessage(
            msg,
            "Passwords do not match.",
            false
        );

        return;

    }


    let users = getUsers();


    if(users.some(u => u.mobile === mobile)){

        showMessage(
            msg,
            "Mobile number already registered.",
            false
        );

        return;

    }


    if(users.some(u => u.email === email)){

        showMessage(
            msg,
            "Email already registered.",
            false
        );

        return;

    }


    let user = {

        id:"U" + Date.now(),

        name:name,

        mobile:mobile,

        email:email,

        password:password

    };


    users.push(user);


    localStorage.setItem(
        "infraUsers",
        JSON.stringify(users)
    );


    showMessage(
        msg,
        "Registration successful. Please login.",
        true
    );


    document.getElementById("regName").value="";
    document.getElementById("regMobile").value="";
    document.getElementById("regEmail").value="";
    document.getElementById("regPassword").value="";
    document.getElementById("regConfirm").value="";


    setTimeout(function(){

        showLogin();

        document.getElementById("loginId").value =
            mobile;

    },1000);

}


/* =================================================
   LOGIN
================================================= */

function login(){

    let type =
        document.getElementById("loginType").value;

    let id =
        document.getElementById("loginId").value.trim().toLowerCase();

    let password =
        document.getElementById("loginPass").value;

    let msg =
        document.getElementById("loginMessage");


    if(type === "admin"){

        if(id === "admin" && password === "admin123"){

            document.getElementById("auth").style.display="none";
            document.getElementById("adminApp").style.display="block";

            updateAdminDashboard();

            return;

        }

        showMessage(
            msg,
            "Invalid admin username or password.",
            false
        );

        return;

    }


    let users = getUsers();


    let user = users.find(function(u){

        return (
            (u.mobile === id || u.email === id) &&
            u.password === password
        );

    });


    if(!user){

        showMessage(
            msg,
            "Invalid citizen login details.",
            false
        );

        return;

    }


    localStorage.setItem(
        "currentUser",
        JSON.stringify(user)
    );


    document.getElementById("auth").style.display="none";
    document.getElementById("citizenApp").style.display="block";


    loadCitizenProfile();

}

    element.className =

        ? "message success"
        : "message error";

    element.innerText=text;

}

/* =================================================
   GPS
================================================= */

function getLocation(){

    if(!navigator.geolocation){

        alert("Geolocation is not supported.");

        return;

    }


    navigator.geolocation.getCurrentPosition(

        function(position){

            let lat =
                position.coords.latitude;

            let lng =
                position.coords.longitude;


            document.getElementById("lat")
                .value=lat;

            document.getElementById("lng")
                .value=lng;


            document.getElementById("complaintLocation")
                .value=
                "Latitude: "+lat.toFixed(6)+
                ", Longitude: "+lng.toFixed(6);

        },

        function(error){

            alert(
                "Location permission was denied. Please allow location access in Chrome."
            );

        }

    );

}


/* =================================================
   DEPARTMENT
================================================= */

function getDepartment(type){

    if(type==="Road")
        return "Road & Transport Department";

    if(type==="Bridge")
        return "Public Works Department (PWD)";

    if(type==="Streetlight")
        return "Electricity / Municipal Department";

    if(type==="Water Pipeline")
        return "Water Supply Department";

    if(type==="Drainage")
        return "Municipal / Drainage Department";

    if(type==="Public Building")
        return "Public Works Department (PWD)";

    return "Municipal Authority";

}


/* =================================================
   RISK
================================================= */

function getRisk(severity){

    if(severity==="Critical")
        return 95;

    if(severity==="High")
        return 80;

    if(severity==="Medium")
        return 55;

    return 25;

}


/* =================================================
   SUBMIT COMPLAINT
================================================= */

function submitComplaint(){

    let user=getCurrentUser();

    if(!user){

        alert("Please login.");

        return;

    }


    let type =
        document.getElementById("infraType").value;

    let severity =
        document.getElementById("severity").value;

    let location =
        document.getElementById("complaintLocation").value;

    let description =
        document.getElementById("description").value.trim();

    let lat =
        document.getElementById("lat").value;

    let lng =
        document.getElementById("lng").value;


    if(!type || !severity || !location){

        alert(
            "Please select infrastructure, severity and detect location."
        );

        return;

    }


    let complaint={

        id:"REQ-"+Date.now(),

        userId:user.id,

        citizen:user.name,

        mobile:user.mobile,

        email:user.email,

        infrastructure:type,

        severity:severity,

        location:location,

        latitude:lat,

        longitude:lng,

        description:description,

        department:getDepartment(type),

        risk:getRisk(severity),

        status:"Pending",

        date:new Date().toLocaleString()

    };


    let complaints=getComplaints();

    complaints.unshift(complaint);

    saveComplaints(complaints);


    let msg =
        document.getElementById("submitMessage");


    showMessage(
        msg,
        "Request submitted successfully. Request ID: "+
        complaint.id,
        true
    );


    document.getElementById("infraType").value="";
    document.getElementById("severity").value="";
    document.getElementById("complaintLocation").value="";
    document.getElementById("description").value="";
    document.getElementById("lat").value="";
    document.getElementById("lng").value="";


    updateCitizenDashboard();

}

/* =================================================
   ADMIN REQUESTS
================================================= */

function loadAdminRequests(){

    let all=getComplaints();


    let search =
        document.getElementById("searchRequest")
        .value
        .toLowerCase();


    let status =
        document.getElementById("filterStatus")
        .value;


    let department =
        document.getElementById("filterDepartment")
        .value;


    let filtered=all.filter(function(c){

        let matchesSearch =
            !search ||
            c.id.toLowerCase().includes(search) ||
            c.citizen.toLowerCase().includes(search) ||
            c.mobile.includes(search);


        let matchesStatus =
            status==="All" ||
            c.status===status;


        let matchesDepartment =
            department==="All" ||
            c.department===department;


        return (
            matchesSearch &&
            matchesStatus &&
            matchesDepartment
        );

    });


    let box=
        document.getElementById("adminRequestList");


    if(filtered.length===0){

        box.innerHTML=
        '<div class="empty">No requests found.</div>';

        return;

    }


    box.innerHTML=filtered.map(function(c){

        let riskClass =
            c.risk>=80
            ? "critical"
            : c.risk>=60
            ? "high"
            : c.risk>=40
            ? "medium"
            : "low";


        let statusClass =
            c.status==="Resolved"
            ? "resolved"
            : c.status==="In Progress"
            ? "progress"
            : "pending";


        return `

        <div class="complaint">

            <h3>
                ${c.id}
            </h3>

            <p>
                <b>👤 Citizen:</b>
                ${c.citizen}
            </p>

            <p>
                <b>📱 Mobile:</b>
                ${c.mobile}
            </p>

            <p>
                <b>📧 Email:</b>
                ${c.email}
            </p>

            <p>
                <b>🏗 Infrastructure:</b>
                ${c.infrastructure}
            </p>

            <p>
                <b>🏢 Department:</b>
                ${c.department}
            </p>

            <p>
                <b>⚠ Severity:</b>
                <span class="badge ${riskClass}">
                    ${c.severity}
                </span>
            </p>

            <p>
                <b>🤖 AI Risk Score:</b>
                ${c.risk}/100
            </p>

            <p>
                <b>📍 Location:</b>
            </p>

            <div class="location">
                ${c.location}
                <br>
                Latitude: ${c.latitude || "N/A"}
                <br>
                Longitude: ${c.longitude || "N/A"}
            </div>

            <p>
                <b>📝 Description:</b>
                ${c.description || "No description"}
            </p>

            <p>
                <b>📅 Submitted:</b>
                ${c.date}
            </p>

            <p>
                <b>Current Status:</b>

                <span class="badge ${statusClass}">

                </span>

            </p>

            <button
                class="small-btn blue-btn"
                onclick="changeStatus('${c.id}','In Progress')">
                🔧 In Progress
            </button>

                class="small-btn green-btn"
                onclick="changeStatus('${c.id}','Resolved')">
                ✅ Resolve
            </button>

            <button
                class="small-btn red-btn"
                onclick="deleteRequest('${c.id}')">
                🗑 Delete
            </button>

        </div>

        `;

    }).join("");

}


/* =================================================
   CHANGE STATUS
================================================= */

function changeStatus(id,status){

    let all=getComplaints();


    let index=all.findIndex(
        c => c.id===id
    );


    if(index===-1){

        return;

    }


    all[index].status=status;


    saveComplaints(all);


    loadAdminRequests();

    updateAdminDashboard();

}


/* =================================================
   DELETE REQUEST
================================================= */

function deleteRequest(id){

    if(!confirm(
        "Are you sure you want to delete this request?"
    )){

        return;

    }


    let all=getComplaints();


    all=all.filter(
        c => c.id!==id
    );


    saveComplaints(all);


    loadAdminRequests();

    updateAdminDashboard();

}
